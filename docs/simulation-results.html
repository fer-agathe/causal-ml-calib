res <- list()

for (i_grid in 1:6) {
  cat(
    sprintf(
      "Load results from simulation (%d/%d)...\n\n",
      i_grid, nrow(grid)
    )
  )
  file_path <- str_c(output_dir, i_grid, ".rda")

  if (!file.exists(file_path)) next

  load(file_path) # Loads 'res_tmp'

  for (j in seq_along(res_tmp)) {
    # Extract DGP parameters
    cur_dgp  <- res_tmp[[j]]$dgp_params
    cur_clip <- res_tmp[[j]]$clipping_threshold

    # Annotate ATE table
    res_tmp[[j]]$ate <- res_tmp[[j]]$ate |>
      mutate(
        n                  = cur_dgp$n,
        p                  = cur_dgp$p,
        Rd2                = cur_dgp$Rd2,
        Ry2                = cur_dgp$Ry2,
        rho                = cur_dgp$rho,
        clipping_threshold = cur_clip
      )

    # Annotate GATE table
    res_tmp[[j]]$gate <- res_tmp[[j]]$gate |>
      mutate(
        n                  = cur_dgp$n,
        p                  = cur_dgp$p,
        Rd2                = cur_dgp$Rd2,
        Ry2                = cur_dgp$Ry2,
        rho                = cur_dgp$rho,
        clipping_threshold = cur_clip
      )

    # Annotate Metrics table
    res_tmp[[j]]$metrics <- res_tmp[[j]]$metrics |>
      mutate(
        n                  = cur_dgp$n,
        p                  = cur_dgp$p,
        Rd2                = cur_dgp$Rd2,
        Ry2                = cur_dgp$Ry2,
        rho                = cur_dgp$rho,
        clipping_threshold = cur_clip
      )
  }

  res <- c(res, res_tmp)
}

#' Format numbers
#'
#' |x| >= 100: 0 digit;
#' |x| >= 10: 1 digit;
#' otherwise: 3 digits
fmt_num <- function(x) {
  dplyr::case_when(
    abs(x) >= 100 ~ sprintf("%.0f", x),
    abs(x) >= 10  ~ sprintf("%.1f", x),
    TRUE          ~ sprintf("%.3f", x)
  )
}

get_table_mc <- function(n_val, p_val, rd2_val, estimator) {
  
  table_data <- res_ate |>
    filter(
      n == n_val,
      p == p_val,
      Rd2 == rd2_val
    ) |>
    mutate(
      calib_name = factor(
        calib_name, 
        levels = c("uncalibrated", "isotonic", "platt", "uncalib-platt-clip", "beta", "uncalib-beta-clip", "joint-platt"),
        labels = c("Uncalibrated", "Isotonic", "Platt", "Uncalibrated Platt-clip", "Beta", "Uncalibrated Beta-clip", "Outcome-informed")
      ),
      g_name = factor(
        g_name,
        levels = c("ols", "lasso", "rf", "lgbm"),
        labels = c("Outcome: OLS", "Outcome: Lasso", "Outcome: Random Forest", "Outcome: Boosting")
      ),
      m_name = factor(
        m_name,
        levels = c("logit", "rf_classif", "rf_reg", "lgbm"),
        labels = c("Logit", "RF Classif", "RF Reg", "Boosting")
      )
    ) |> 
    group_by(g_name, m_name, calib_name) |>
    mutate(
      estim_ate = .data[[paste0("ate_", estimator)]]
    ) |>
    summarise(
      bias = mean(estim_ate - ate_true),
      sd   = sd(estim_ate - ate_true),
      .groups = "drop"
    ) |>
    pivot_longer(
      cols = c(bias, sd),
      names_to = "metric",
      values_to = "value"
    ) |>
    unite(
      col = "column",
      calib_name,
      metric,
      sep = "__"
    ) |>
    pivot_wider(
      names_from = column,
      values_from = value
    )
  
  formatted_table <- table_data |>
    gt(
      rowname_col = "m_name",
      groupname_col = "g_name"
    ) |>
    tab_spanner_delim(
      delim = "__",
      columns = contains("__")
    ) |>
    cols_label(
      .list = setNames(
        ifelse(
          endsWith(names(table_data), "__bias"),
          "Bias",
          ifelse(
            endsWith(names(table_data), "__sd"),
            "(SD)",
            names(table_data)
          )
        ),
        names(table_data)
      )
    ) |>
    fmt_number(
      columns = ends_with("__bias"),
      decimals = 3
    ) |>
    fmt_number(
      columns = ends_with("__sd"),
      decimals = 3,
      pattern = "({x})"
    ) |>
    cols_align(
      align = "center",
      columns = contains("__")
    ) |>
    tab_header(
      title = md("**Monte Carlo Simulation Results**"),
      subtitle = md(
        glue("Estimator: **{toupper(estimator)}** | *N* = {n_val}, *P* = {p_val}, Rd2 = {rd2_val}")
      )
    ) |>
    opt_stylize(style = 1, color = "gray") |>
    tab_options(
      column_labels.font.weight = "bold",
      row_group.font.weight = "bold"
    )
  
  formatted_table
}

      n == n_val,
      p == p_val,
      Rd2 == rd2_val,
      calib_name == !!calib
    ) |> 
    mutate(
      estim_ate = !!sym(estimator)
    ) |>
    group_by(g_name, m_name) |> 
    summarise(
      bias = mean(estim_ate - ate_true),
      sd   = sd(estim_ate - ate_true),
      .groups = "drop"
    ) |> 
    pivot_longer(
      cols = c(bias, sd),
      names_to = "metric",
      values_to = "value"
    ) |> 
    mutate(
      column = !!lab
    ) |> 
    unite(
      col = "column",
      column,
      metric,
      sep = "__"
    ) |> 
    pivot_wider(
      names_from = column,
      values_from = value
    )
}
